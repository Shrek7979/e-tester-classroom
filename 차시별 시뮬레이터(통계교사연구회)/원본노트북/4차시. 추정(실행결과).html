5. 코드 셀 5
import numpy as np
import matplotlib.pyplot as plt
import scipy.stats as stats
import matplotlib.font_manager as fm

# 모집단 매개변수 (참값)
population_mean_true = 500
population_std = 20

# 시뮬레이션 매개변수
sample_size = 25
n_intervals = 100
confidence_level = 0.95

# z-스코어 계산 (95% 신뢰구간)
z_score = stats.norm.ppf(1 - (1 - confidence_level) / 2)

# 신뢰구간 저장 리스트
confidence_intervals = []

# 참값을 포함하는 구간 카운트
covered_count = 0

for i in range(n_intervals):
    # 모집단에서 표본 추출
    sample = np.random.normal(loc=population_mean_true, scale=population_std, size=sample_size)

    # 표본평균 계산
    sample_mean = np.mean(sample)

    # 표준오차 계산
    standard_error = population_std / np.sqrt(sample_size)

    # 신뢰구간 계산
    margin_of_error = z_score * standard_error
    lower_bound = sample_mean - margin_of_error
    upper_bound = sample_mean + margin_of_error

    confidence_intervals.append((lower_bound, upper_bound))

    # 참값 포함 여부 확인
    if lower_bound <= population_mean_true <= upper_bound:
        covered_count += 1

# 폰트 설정 (히스토그램 셀과 동일)
# 나눔 고딕 폰트 설치 및 설정
plt.rc('font', family='NanumGothic')
plt.rcParams['axes.unicode_minus'] = False # 마이너스 폰트 깨짐 방지

# 시각화
plt.figure(figsize=(14, 10))

for i, (lower, upper) in enumerate(confidence_intervals):
    color = 'blue' if lower <= population_mean_true <= upper else 'red'
    plt.plot([lower, upper], [i, i], color=color, linewidth=2)

plt.axvline(x=population_mean_true, color='black', linestyle='--', label=f'모평균 참값 ({population_mean_true}시간)')

plt.title(f'{n_intervals}개 표본에 대한 95% 신뢰구간 (표본 크기={sample_size})')
plt.xlabel('신뢰구간')
plt.ylabel('표본 번호')
plt.yticks([]) # y축 눈금 제거
plt.legend()
plt.grid(True, linestyle='--', alpha=0.6)
plt.show()

# 결과 출력
print(f"\n전체 {n_intervals}개 구간 중 모평균 참값 {population_mean_true}를 포함한 구간 수: {covered_count}개")
print(f"모평균 참값 포함 비율: {covered_count / n_intervals:.2%} (이론적 기대치: {confidence_level:.0%})")

import numpy as np
import matplotlib.pyplot as plt
import scipy.stats as stats
import pandas as pd
import matplotlib.font_manager as fm

# 폰트 설정 (이전 셀에서 설치했으므로 재설치 스킵, 설정만)
plt.rc('font', family='NanumGothic')
plt.rcParams['axes.unicode_minus'] = False # 마이너스 폰트 깨짐 방지

# 모집단 매개변수
population_std = 20
confidence_level = 0.95

# Z-스코어 계산 (95% 신뢰도)
z_score = stats.norm.ppf(1 - (1 - confidence_level) / 2)

# 다양한 표본 크기
sample_sizes = np.array([10, 25, 50, 100, 200, 400, 800])

# 오차 한계 계산
margin_of_errors = z_score * (population_std / np.sqrt(sample_sizes))

# 결과 표 생성
results_data = {
    '표본 크기 (n)': sample_sizes,
    '오차 한계': margin_of_errors
}
margin_df = pd.DataFrame(results_data)
print("95% 신뢰도에서의 표본 크기별 오차 한계:")
display(margin_df)

# 오차 한계가 2 이하가 되는 표본 크기 찾기
sample_size_for_error_2 = None
for n, moe in zip(sample_sizes, margin_of_errors):
    if moe <= 2:
        sample_size_for_error_2 = n
        break

print(f"\n오차 한계가 처음으로 2 이하가 되는 표본 크기: {sample_size_for_error_2}")

# 오차 한계 그래프 시각화
plt.figure(figsize=(10, 6))
plt.plot(sample_sizes, margin_of_errors, marker='o', linestyle='-', color='skyblue')
plt.axhline(y=2, color='red', linestyle='--', label='오차 한계 = 2')
plt.title('표본 크기에 따른 오차 한계 변화 (95% 신뢰도)')
plt.xlabel('표본 크기 (n)')
plt.ylabel('오차 한계')
plt.grid(True, linestyle='--', alpha=0.7)
plt.legend()
plt.show()

import numpy as np

# 표본 1개당 측정 비용 (5만원)
cost_per_sample = 50000

# 총 비용 계산
total_costs = sample_sizes * cost_per_sample

# 비용과 오차 한계 그래프
fig, ax1 = plt.subplots(figsize=(12, 7))

color = 'tab:blue'
ax1.set_xlabel('표본 크기 (n)')
ax1.set_ylabel('총 비용 (만원)', color=color)
ax1.plot(sample_sizes, total_costs / 10000, color=color, marker='o', label='총 비용') # 만원 단위로 표시
ax1.tick_params(axis='y', labelcolor=color)
ax1.grid(True, linestyle='--', alpha=0.6)

# 두 번째 y축 생성 (오차 한계)
ax2 = ax1.twinx()
color = 'tab:red'
ax2.set_ylabel('오차 한계', color=color)
ax2.plot(sample_sizes, margin_of_errors, color=color, marker='x', linestyle='--', label='오차 한계')
ax2.tick_params(axis='y', labelcolor=color)

# 제목 및 범례
fig.suptitle('표본 크기에 따른 총 비용 및 오차 한계 (비용 대비 정밀도)', fontsize=16)
fig.legend(loc="upper right", bbox_to_anchor=(1,1), bbox_transform=ax1.transAxes)
