import numpy as np
import matplotlib.pyplot as plt

# 시뮬레이션 설정
num_tosses = 10000
num_simulations = 5

# 결과를 저장할 리스트
results = []

for _ in range(num_simulations):
    # 동전 던지기 시뮬레이션 (0: 뒷면, 1: 앞면)
    tosses = np.random.randint(0, 2, num_tosses)

    # 누적 앞면 개수 계산
    cumulative_heads = np.cumsum(tosses)

    # 누적 상대도수 계산
    cumulative_relative_frequency = cumulative_heads / (np.arange(1, num_tosses + 1))
    results.append(cumulative_relative_frequency)

print("시뮬레이션 데이터 준비 완료!")

import numpy as np
import matplotlib.pyplot as plt
import pandas as pd

# 시뮬레이션 설정
ns = [60, 600, 6000]  # 주사위 던지는 횟수
num_experiments = 2000 # 각 n에 대해 반복할 실험 횟수

# 이론적 확률 (6의 눈이 나올 확률)
p_theoretical = 1/6

# 결과를 저장할 딕셔너리
relative_frequencies = {}

for n in ns:
    # 각 n에 대해 num_experiments번 실험 반복
    experiment_results = []
    for _ in range(num_experiments):
        # 주사위 n번 던지기 (1부터 6까지)
        rolls = np.random.randint(1, 7, n)
        # 6의 눈이 나온 횟수
        count_sixes = np.sum(rolls == 6)
        # 상대도수 계산
        relative_freq = count_sixes / n
        experiment_results.append(relative_freq)
    relative_frequencies[n] = np.array(experiment_results)

print("주사위 던지기 시뮬레이션 데이터 준비 완료!")

import numpy as np
import matplotlib.pyplot as plt
import pandas as pd

# 전설 등급 참확률
p_true = 0.05

# 시행 횟수 n 값들
ns_gacha = np.array([100, 500, 1000, 1900, 5000, 7600, 20000])

# 결과를 저장할 리스트
summary_data_gacha = []

print("시행 횟수에 따른 표준편차 및 오차한계 계산...")

for n in ns_gacha:
    # 상대도수의 이론적 표준편차 (이항분포의 표준편차 / n)
    # 이항분포의 표준편차 = sqrt(n * p * (1-p))
    # 상대도수의 표준편차 = sqrt(p * (1-p) / n)
    std_dev_relative_freq = np.sqrt(p_true * (1 - p_true) / n)

    # 오차한계 (표준편차의 2배)
    margin_of_error = 2 * std_dev_relative_freq

    summary_data_gacha.append({
        'n (시행 횟수)': n,
        '상대도수 표준편차': f'{std_dev_relative_freq:.6f}',
        '오차한계 (2*표준편차)': f'{margin_of_error:.6f}'
    })

df_gacha_summary = pd.DataFrame(summary_data_gacha)
df_gacha_summary = df_gacha_summary.set_index('n (시행 횟수)')

print("\n시행 횟수에 따른 상대도수 표준편차 및 오차한계:")
display(df_gacha_summary)

13. 5. 참확률 0.05인 뽑기를 1000번 하는 실험을 2000번 반복해, 전설이 30번 이하로 나오는 경우의 비율을 계산해 출
# 5. 참확률 0.05인 뽑기를 1000번 하는 실험을 2000번 반복해, 전설이 30번 이하로 나오는 경우의 비율을 계산해 출력해줘.

num_draws_per_experiment = 1000 # 한 실험당 뽑기 횟수
num_total_experiments = 2000 # 총 실험 반복 횟수
threshold_legendary_count = 30 # 전설 등급이 30번 이하로 나오는 경우

# 결과를 저장할 리스트
legendary_counts = []

print(f"\n참확률 {p_true}인 뽑기를 {num_draws_per_experiment}번 하는 실험을 {num_total_experiments}번 반복 중...")

for _ in range(num_total_experiments):
    # 이항 분포를 사용하여 전설 등급 아이템 개수 시뮬레이션
    # np.random.binomial(n, p, size=1)은 n번의 시행 중 p확률로 성공하는 횟수를 반환
    num_legendary_items = np.random.binomial(num_draws_per_experiment, p_true, size=1)[0]
    legendary_counts.append(num_legendary_items)

# 전설이 30번 이하로 나오는 경우의 수 계산
count_below_threshold = np.sum(np.array(legendary_counts) <= threshold_legendary_count)

# 해당 경우의 비율 계산
proportion_below_threshold = count_below_threshold / num_total_experiments

print(f"전설 등급이 {threshold_legendary_count}번 이하로 나온 경우의 비율: {proportion_below_threshold:.4f}")
